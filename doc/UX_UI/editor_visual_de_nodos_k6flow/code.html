import http from 'k6/http';
import { check, sleep } from 'k6';

// Opciones de ejecución generadas automáticamente
export const options = {
  stages: [
    { duration: '2m', target: 50 },   // Rampa de calentamiento inicial
    { duration: '5m', target: 200 },  // Carga sostenida peak
    { duration: '1m', target: 0 },    // Descenso controlado
  ],
  thresholds: {
    http_req_duration: ['p(95)<200'], // 95% de peticiones bajo 200ms
    http_req_failed: ['rate<0.01'],    // Menos del 1% de errores
  },
};

export default function () {
  // Nodo: Autenticación JWT
  const authRes = http.post('https://api.pagos.internal/auth/token', JSON.stringify({
    client_id: 'loadtest-runner',
    secret: 'k6-session-sec'
  }), { headers: { 'Content-Type': 'application/json' } });
  
  const token = authRes.json('token_auth');

  // Nodo: GET /api/v1/checkout/pedidos (Seleccionado)
  const res = http.get(`https://api.pagos.internal/v1/checkout/pedidos?session=${__VU}`, {
    headers: {
      'Authorization': `Bearer ${token}`,
      'Content-Type': 'application/json',
      'X-Request-Trace': `${__ITER}-${Date.now()}`
    },
    timeout: '15000ms'
  });

  // Validación de umbrales
  check(res, {
    'status is 200': (r) => r.status === 200,
    'latencia aceptable': (r) => r.timings.duration < 250,
  });

  sleep(1);
}
